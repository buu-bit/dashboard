  <script>

    function updateClock() {

      const now = new Date();

      document.getElementById('clock').textContent = now.toTimeString().split(' ')[0];

      const days = ['日', '一', '二', '三', '四', '五', '六'];

      document.getElementById('date').textContent = `${now.getFullYear()}年${now.getMonth()+1}月${now.getDate()}日 星期${days[now.getDay()]}`;

    }

    setInterval(updateClock, 1000);

    updateClock();


    async function fetchWeather() {

      try {

        const res = await fetch('https://data.weather.gov.hk/weatherAPI/opendata/weather.php?dataType=rhrread&lang=tc');

        const data = await res.json();

        if(data.temperature && data.temperature.data) {

          document.getElementById('temp').textContent = data.temperature.data[0].value;

          const tw = data.temperature.data.find(t => t.place.includes('荃灣'));

          if(tw) document.getElementById('region-temp').textContent = `${tw.value}°C`;

        }

        if(data.humidity && data.humidity.data) {

          document.getElementById('humidity').textContent = `${data.humidity.data[0].value}%`;

        }

      } catch(e) {}

    }

    fetchWeather();

    setInterval(fetchWeather, 300000);


    async function fetchKMB(route, bound, elementId) {

      const container = document.getElementById(elementId);

      try {

        const res = await fetch(`https://data.etabus.gov.hk/v1/transport/kmb/route-eta/${route}/1`);

        const data = await res.json();

        const now = new Date();

        const etas = data.data

          .filter(i => i.dir === bound && i.eta)

          .map(i => Math.round((new Date(i.eta) - now)/60000))

          .filter(m => m >= 0)

          .sort((a,b) => a-b)

          .slice(0, 3);

        

        if(etas.length === 0) {

          container.innerHTML = '<div class="eta-item"><div class="eta-time">--</div><div class="eta-label">暫無班次</div></div>';

        } else {

          container.innerHTML = etas.map(m => `

            <div class="eta-item">

              <div class="eta-time">${m === 0 ? '即將到' : m + ' 分鐘'}</div>

              <div class="eta-label">預計</div>

            </div>

          `).join('');

        }

      } catch(e) {

        container.innerHTML = '<div class="eta-item"><div class="eta-time">--</div></div>';

      }

    }

    }
